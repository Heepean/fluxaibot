Пример подключения (Python)

import openai
client = openai.OpenAI(
    base_url="http://127.0.0.1:12345/v1",
    api_key="hub-token"
)
response = client.chat.completions.create(
    model="local-model",
    messages=[{"role": "user", "content": "Привет!"}]
)
